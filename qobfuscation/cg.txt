    <script>
        const gamesElement = document.getElementById("games");
        const statusElement = document.getElementById("status");
        const searchElement = document.getElementById("search");
        const playerPage = document.getElementById("playerPage");
        const playerTitle = document.getElementById("playerTitle");
        const loadingElement = document.getElementById("loading");
        const loadingVideo = document.getElementById("loadingVideo");
        const loadingMessage = document.getElementById("loadingMessage");
        const closePlayer = document.getElementById("closePlayer");
        const fullscreen = document.getElementById("fullscreen");

        const sessionKey = "celestial-cg-session";
        const cooldownKey = "celestial-cg-cooldown";

        const sessionLimit = 45 * 60 * 1000;
        const cooldownLimit = 5 * 60 * 1000;
        const closeCooldownLimit = 90 * 1000;

        const tabId = crypto.randomUUID();

        let lessons = [];
        let currentGame = null;
        let sessionTimer = null;
        let endingForLimit = false;

        const astra = new Astra({
            mount: "#player"
        });

        function getActiveSession() {
            try {
                const value = localStorage.getItem(sessionKey);

                if (!value) {
                    return null;
                }

                const session = JSON.parse(value);

                if (
                    !session ||
                    !session.startedAt ||
                    Date.now() - session.startedAt >= sessionLimit
                ) {
                    localStorage.removeItem(sessionKey);
                    return null;
                }

                return session;
            } catch {
                localStorage.removeItem(sessionKey);
                return null;
            }
        }

        function getCooldownRemaining() {
            const value = Number(
                localStorage.getItem(cooldownKey)
            );

            if (!value) {
                return 0;
            }

            const remaining =
                value - Date.now();

            if (remaining <= 0) {
                localStorage.removeItem(cooldownKey);
                return 0;
            }

            return remaining;
        }

        function formatTime(milliseconds) {
            const seconds = Math.ceil(
                milliseconds / 1000
            );

            const minutes = Math.floor(
                seconds / 60
            );

            const remainingSeconds =
                seconds % 60;

            return `${minutes}:${String(
                remainingSeconds
            ).padStart(2, "0")}`;
        }

        function startCooldown(duration = cooldownLimit) {
            localStorage.setItem(
                cooldownKey,
                String(
                    Date.now() + duration
                )
            );
        }

        function claimSession(game) {
            const active = getActiveSession();

            if (
                active &&
                active.tabId !== tabId
            ) {
                return false;
            }

            localStorage.setItem(
                sessionKey,
                JSON.stringify({
                    tabId,
                    game: game.gameName,
                    startedAt: Date.now()
                })
            );

            return true;
        }

        function releaseSession() {
            const active = getActiveSession();

            if (
                active &&
                active.tabId === tabId
            ) {
                localStorage.removeItem(
                    sessionKey
                );
            }

            if (sessionTimer) {
                clearTimeout(sessionTimer);
                sessionTimer = null;
            }
        }

        function resetPlayer() {
            currentGame = null;

            playerPage.classList.remove(
                "open"
            );

            loadingElement.classList.remove(
                "show"
            );

            loadingElement.classList.remove("timed-out");

            loadingVideo.pause();
            loadingVideo.currentTime = 0;
            loadingVideo.muted = false;

            playerTitle.textContent = "";
        }

        async function endForLimit() {
            if (!currentGame) {
                return;
            }

            endingForLimit = true;

            startCooldown(
                cooldownLimit
            );

            releaseSession();

            try {
                await astra.end();
            } catch {
                astra.stop();
            }

            resetPlayer();

            alert(
                "your 45 minute session has ended. wait 5 minutes before playing again."
            );

            endingForLimit = false;
        }

        function startSessionTimer() {
            if (sessionTimer) {
                clearTimeout(
                    sessionTimer
                );
            }

            const active =
                getActiveSession();

            if (!active) {
                return;
            }

            const remaining =
                sessionLimit -
                (
                    Date.now() -
                    active.startedAt
                );

            sessionTimer = setTimeout(
                endForLimit,
                Math.max(
                    0,
                    remaining
                )
            );
        }

        function renderGames(games) {
            gamesElement.innerHTML = "";

            if (!games.length) {
                statusElement.textContent =
                    "no games found.";

                statusElement.style.display =
                    "block";

                return;
            }

            statusElement.style.display =
                "none";

            games.forEach(game => {
                const card =
                    document.createElement(
                        "button"
                    );

                const image =
                    document.createElement(
                        "img"
                    );

                const info =
                    document.createElement(
                        "div"
                    );

                const name =
                    document.createElement(
                        "div"
                    );

                const provider =
                    document.createElement(
                        "div"
                    );

                card.className = "game";

                image.src = game.logo;
                image.alt = game.gameName;
                image.loading = "lazy";

                info.className =
                    "game-info";

                name.className =
                    "game-name";

                name.textContent =
                    game.gameName;

                provider.className =
                    "game-provider";

                provider.textContent =
                    game.provider;

                info.appendChild(name);
                info.appendChild(provider);

                card.appendChild(image);
                card.appendChild(info);

                card.addEventListener(
                    "click",
                    () => {
                        playGame(game);
                    }
                );

                gamesElement.appendChild(
                    card
                );
            });
        }

        async function loadLessons() {
            try {
                const response =
                    await fetch(
                        "/cg/lessons.json",
                        {
                            cache: "no-store"
                        }
                    );

                if (!response.ok) {
                    throw new Error(
                        "failed to load lessons"
                    );
                }

                const data =
                    await response.json();

                if (
                    !data.ok ||
                    !Array.isArray(
                        data.games
                    )
                ) {
                    throw new Error(
                        "invalid lessons json"
                    );
                }

                lessons = data.games;

                renderGames(
                    lessons
                );
            } catch (error) {
                console.error(error);

                statusElement.textContent =
                    "failed to load games.";
            }
        }

        async function playGame(game) {
            const cooldown =
                getCooldownRemaining();

            if (cooldown > 0) {
                alert(
                    `you need to wait ${formatTime(
                        cooldown
                    )} before playing again.`
                );

                return;
            }

            const active =
                getActiveSession();

            if (
                active &&
                active.tabId !== tabId
            ) {
                alert(
                    `${active.game} is already running in another tab.`
                );

                return;
            }

            if (currentGame) {
                alert(
                    `${currentGame.gameName} is already running.`
                );

                return;
            }

            if (!claimSession(game)) {
                alert(
                    "cloud gaming is already running in another tab."
                );

                return;
            }

            currentGame = game;

            playerTitle.textContent =
                game.gameName;

            playerPage.classList.add(
                "open"
            );

            loadingElement.classList.remove(
                "timed-out"
            );

            loadingElement.classList.add(
                "show"
            );

            loadingVideo.currentTime = 0;
            loadingVideo.muted = false;

            loadingVideo.play().catch(() => {
                loadingVideo.muted = true;
                loadingVideo.play().catch(() => {});
            });

            try {
                const response =
                    await fetch(
                        "/api/cg",
                        {
                            method: "POST",
                            headers: {
                                "Content-Type":
                                    "application/json"
                            },
                            body:
                                JSON.stringify({
                                    lesson:
                                        game.embedId
                                })
                        }
                    );

                const session =
                    await response.json();
                console.log("astra session:", session);

                if (!response.ok) {
                    throw new Error(
                        session.detail ||
                        session.error ||
                        "failed to create session"
                    );
                }

                startSessionTimer();

                await astra.play(
                    session
                );
            } catch (error) {
                releaseSession();

                currentGame = null;

                console.error(
                    error
                );

                loadingElement.textContent =
                    error instanceof Error
                        ? error.message
                        : "failed to start game";
            }
        }

        searchElement.addEventListener(
            "input",
            () => {
                const query =
                    searchElement.value
                        .trim()
                        .toLowerCase();

                if (!query) {
                    renderGames(
                        lessons
                    );

                    return;
                }

                const filtered =
                    lessons.filter(
                        game => {
                            return (
                                game.gameName
                                    .toLowerCase()
                                    .includes(
                                        query
                                    ) ||
                                game.provider
                                    .toLowerCase()
                                    .includes(
                                        query
                                    )
                            );
                        }
                    );

                renderGames(
                    filtered
                );
            }
        );

        closePlayer.addEventListener(
            "click",
            async () => {
                if (!currentGame) {
                    return;
                }

                startCooldown(
                    closeCooldownLimit
                );

                releaseSession();

                try {
                    await astra.end();
                } catch {
                    astra.stop();
                }

                resetPlayer();

                alert(
                    "session closed. wait 1:30 before playing again."
                );
            }
        );

        fullscreen.addEventListener(
            "click",
            () => {
                astra.fullscreen();
            }
        );

        astra.on(
            "connecting",
            () => {
                loadingElement.classList.add(
                    "show"
                );
            }
        );

        astra.on(
            "ready",
            () => {
                loadingVideo.muted = true;
                loadingVideo.pause();

                loadingElement.classList.remove(
                    "show"
                );
            }
        );

        astra.on(
            "ended",
            () => {
                releaseSession();

                resetPlayer();

                endingForLimit =
                    false;
            }
        );

        loadingVideo.addEventListener(
            "ended",
            () => {
                if (!loadingElement.classList.contains("show")) {
                    return;
                }

                loadingElement.classList.add(
                    "timed-out"
                );
            }
        );

        const originalFetch = window.fetch.bind(window);

        window.fetch = async (...args) => {
            const response = await originalFetch(...args);
            const target = String(args[0] instanceof Request ? args[0].url : args[0]);

            if (target.includes("/astra/api/coursework/progress/")) {
                response.clone().json().then(data => {
                    if (data && data.phase === "error" && data.error) {
                        loadingVideo.muted = true;
                        loadingVideo.pause();

                        loadingMessage.textContent = data.error;

                        loadingElement.classList.add(
                            "timed-out"
                        );

                        loadingElement.classList.add(
                            "show"
                        );
                    }
                }).catch(() => {});
            }

            return response;
        };

        astra.on(
            "error",
            event => {
                console.error(
                    event
                );

                loadingVideo.muted = true;
                loadingVideo.pause();

                loadingElement.classList.add(
                    "timed-out"
                );

                loadingElement.classList.add(
                    "show"
                );
            }
        );

        window.addEventListener(
            "storage",
            event => {
                if (
                    event.key !==
                    sessionKey
                ) {
                    return;
                }

                const active =
                    getActiveSession();

                if (
                    currentGame &&
                    active &&
                    active.tabId !==
                    tabId
                ) {
                    try {
                        astra.stop();
                    } catch { }

                    releaseSession();

                    resetPlayer();

                    alert(
                        "cloud gaming was opened in another tab."
                    );
                }
            }
        );

        window.addEventListener("pagehide", () => {
            if (!currentGame) {
                return;
            }

            const active = getActiveSession();

            if (
                active &&
                active.tabId === tabId
            ) {
                localStorage.setItem(
                    cooldownKey,
                    String(
                        Date.now() +
                        closeCooldownLimit
                    )
                );

                localStorage.removeItem(
                    sessionKey
                );
            }
        });

        loadLessons();
    </script>
