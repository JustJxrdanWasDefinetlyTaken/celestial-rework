  <script type="module">
    import {
      installLucideWispEncryption,
      LUCIDE_WISP_URL
    } from "/lucide-wisp.mjs";

    const { Controller } = $scramjetController;
    const { defaultConfig } = $scramjet;
    const EpoxyTransport = self.EpoxyTransport.default;

    const form = document.getElementById("searchForm");
    const input = document.getElementById("searchInput");
    const landingPage = document.getElementById("newTabPage");
    const searchStatus = document.getElementById("navigationStatus");
    const proxyFrame = document.getElementById("proxyFrame");
    const proxyStatus = document.getElementById("proxyStatus");

    let scramjet = null;
    let frame = null;
    let pendingUrl = null;

    function setStatus(message, isError = false) {
      proxyStatus.textContent = message;
      proxyStatus.dataset.error = String(isError);
    }

    function getDestination(value) {
      const query = value.trim();

      if (!query) return null;

      const isUrl =
        /^(https?:\/\/|localhost(?::\d+)?(?:\/|$)|[\w-]+(?:\.[\w-]+)+(?:[/:?#][^\s]*)?$)/i.test(query)
        && !/\s/.test(query);

      if (isUrl) {
        return /^https?:\/\//i.test(query)
          ? query
          : `https://${query}`;
      }

      return `https://www.google.com/search?q=${encodeURIComponent(query)}`;
    }

    function navigate(url) {
      pendingUrl = url;

      if (!scramjet) {
        searchStatus.textContent = "Starting secure browser...";
        setStatus("Starting Scramjet...");
        return;
      }

      try {
        if (!frame) {
          frame = scramjet.createFrame(proxyFrame);
        }

        // Load the requested URL through Scramjet.
        frame.go(url);

        proxyFrame.classList.add("active");
        landingPage.hidden = true;

        searchStatus.textContent = "";
        setStatus("");
      } catch (error) {
        console.error("Scramjet navigation failed:", error);
        setStatus("Could not open that page. Please try again.", true);
      }
    }

    form.addEventListener("submit", (event) => {
      event.preventDefault();

      const destination = getDestination(input.value);

      if (destination) {
        navigate(destination);
      }
    });

    window.addEventListener("load", () => {
      input.focus({ preventScroll: true });
    });

    const autofill = new URLSearchParams(
      window.location.search
    ).get("autofill");

    if (autofill) {
      input.value = autofill;
    }

    async function initializeScramjet() {
      if (!("serviceWorker" in navigator)) {
        throw new Error("This browser does not support service workers.");
      }

      await navigator.serviceWorker.register("/sw.js");

      const registration = await navigator.serviceWorker.ready;
      const serviceworker =
        navigator.serviceWorker.controller ?? registration.active;

      if (!serviceworker) {
        throw new Error("Scramjet service worker is unavailable.");
      }

      installLucideWispEncryption();

      const transport = new EpoxyTransport({
        wisp: LUCIDE_WISP_URL
      });

      await transport.init();

      scramjet = new Controller({
        serviceworker,
        transport,
        scramjetConfig: defaultConfig
      });

      await scramjet.wait();

      setStatus("");

      if (pendingUrl) {
        navigate(pendingUrl);
      }
    }

    initializeScramjet().catch((error) => {
      console.error("Scramjet initialization failed:", error);
      setStatus(error?.message || "Scramjet initialization failed.", true);
    });
  </script>
